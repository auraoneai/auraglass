{/* stories/mat/Tokens.mdx — MAT-333 (REQ-MAT-08/22, TOKENS-THEME-15). Table is
   generated from dist/tokens/manifest.json at build time; no hand-typed
   values. When the manifest has not been generated yet the page reports
   pending instead of blocking (contract §2.4 cross-lane rule). */}
import { Meta } from '@storybook/addon-docs/blocks';
import { globJson, PendingCallout } from './_shared';

export const manifest = globJson(import.meta.glob('/dist/tokens/manifest.json', { eager: true, import: 'default' }));

<Meta title="MAT/Tokens" />

# Tokens

Every row below comes straight from `dist/tokens/manifest.json` (generated by
`npm run build:tokens`). Values are never hand-typed into this page.

{!manifest ? (
  <PendingCallout what="dist/tokens/manifest.json — run `npm run build:tokens` (or wait for lane 2a-T merge)" />
) : (
  <p>
    {manifest.tokens.length} public tokens · version {manifest.version} · generatedFrom{' '}
    <code>{manifest.generatedFrom}</code>
  </p>
)}

{manifest ? (
  <table style={{ borderCollapse: 'collapse', fontFamily: 'ui-monospace, monospace', fontSize: 13 }}>
    <thead>
      <tr>
        {['name', 'tier', 'group', 'value', 'modes', 'swatch'].map((h) => (
          <th key={h} style={{ textAlign: 'left', borderBottom: '2px solid #94a3b8', padding: '4px 10px' }}>{h}</th>
        ))}
      </tr>
    </thead>
    <tbody>
      {manifest.tokens.map((t) => {
        const group = t.name.split('.')[0] ?? '';
        const modes = Object.keys(t.modes ?? {}).join(', ') || '—';
        const isColor = /^(#|rgb|oklch|hsl)/.test(String(t.value));
        return (
          <tr key={t.name}>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>{t.name}</td>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>{t.tier}</td>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>{group}</td>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>{String(t.value)}</td>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>{modes}</td>
            <td style={{ padding: '3px 10px', borderBottom: '1px solid #e2e8f0' }}>
              {isColor ? (
                <span style={{ display: 'inline-flex', alignItems: 'center', gap: 6 }}>
                  <span
                    style={{
                      display: 'inline-block', width: 14, height: 14, borderRadius: 3,
                      background: String(t.value), border: '1px solid #94a3b8',
                    }}
                  />
                  {String(t.value)}
                </span>
              ) : (
                String(t.value)
              )}
            </td>
          </tr>
        );
      })}
    </tbody>
  </table>
) : null}
